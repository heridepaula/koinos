```ts
// src/features/smart-buy/domain/engine-options.ts
/**
 * Opções do motor de aporte (spec koinos-3-1, §3): cada regra do conselho é uma chave, e uma versão
 * nomeada liga as aprovadas. Sem opções, os motores rodam a 3.0 (o comportamento de antes).
 */
export type EngineOptions = {
  /** FIIs: nenhuma ordem abaixo da fatia justa (aporte ÷ nº de FIIs da estratégia). */
  minOrder: "none" | "fairShare";
  /** FIIs: provento suspenso tira o FII do aporte. */
  suspensionBrake: boolean;
  /** Ações: tamanho do lote (1 ou 100, o lote padrão da B3). */
  stockLot: 1 | 100;
  /** Ações: passo de sobra por lote. */
  stockLeftover: boolean;
};

export const ENGINE_VERSIONS = {
  "3.0": { minOrder: "none", suspensionBrake: false, stockLot: 1, stockLeftover: false },
  "3.1": { minOrder: "fairShare", suspensionBrake: true, stockLot: 100, stockLeftover: true },
} as const satisfies Record<string, EngineOptions>;

export type EngineVersion = keyof typeof ENGINE_VERSIONS;

/** A versão que a tela de Recomendações e a coluna Prioridade usam. */
export const CURRENT_ENGINE: EngineVersion = "3.1";
```

  2. `PEDIDO_PEQUENO` (não pede uma cota);
  3. `ABAIXO_DO_MINIMO`, quando `Math.floor(need / price + EPSILON) × price < minimum − EPSILON`, isto é, o próprio pedido não chega ao mínimo;
  4. senão, o motivo de hoje pelo `limitedBy`.
- [ ] **Testes novos** em `describe("sunoCascade: mínimo por ordem")`, todos com `ENGINE_VERSIONS["3.1"]`:
  1. nas 500 carteiras aleatórias, nenhuma ordem tem `total < result.minimum − 0.005`, `spent + leftover = amount` em centavos e só elegíveis compram;

```ts
// prepare(amount, assets, options) devolve também o mínimo
const minimum =
  options.minOrder === "fairShare" && assets.length > 0
    ? round2(Math.min(budget / assets.length, (RISK_CAPS.asset * budget) / 2))
    : 0;

// passo 1
const shares = Math.floor(Math.min(slot.need, allowed) / slot.price + EPSILON);
if (shares > 0 && shares * slot.price >= minimum - EPSILON) buy(slot, shares, false);
if (allowed < slot.need) slot.limitedBy = binding;

// passo 2
const unit = (slot: Slot) =>
  slot.bought > 0 ? 1 : Math.max(1, Math.ceil((minimum - EPSILON) / slot.price));
```

```ts
const SUSPENSION_DAYS = 45;
const SUSPENSION_MIN_PAYMENTS = 6;
const SUSPENSION_MAX_DAYS = 365;

/** Provento suspenso (spec koinos-3-1, §4.2): pagador regular há mais de 45 dias sem data ex, ou um ano sem pagar. */
export function isDividendSuspended(asset: CascadeAsset, options: EngineOptions): boolean {
  const days = asset.daysSinceLastDividend;
  if (!options.suspensionBrake || typeof days !== "number") return false;
  const regular = (asset.dividendPayments12m ?? 0) >= SUSPENSION_MIN_PAYMENTS;
  return (days > SUSPENSION_DAYS && regular) || days > SUSPENSION_MAX_DAYS;
}

function eligibilityFailure(asset: CascadeAsset, options: EngineOptions): EligibilityReason | null {
  // ...as regras de hoje, até AGUARDAR
  if (isDividendSuspended(asset, options)) return "PROVENTO_SUSPENSO";
  return null;
}
```

    2. roda `sunoCascade(fiiPart, fiis, options)` e `stockSmartBuy(stockPart, stocks, options)`;
    3. soma as quantidades compradas.

    Uma parte zero dá resultado `null`.
  - **Sem `prices`:** os preços ficam parados.
  - **`randomPricePaths`:** LCG com semente e Box-Muller. O fator do período `p` é `exp(Σ_{k≤p} (comum_k + próprio_k,ticker))`, a mesma semente dá os mesmos fatores, e um ticker desconhecido tem fator 1.
- [ ] **`scorecard.ts`** exporta `Scorecard` e `scorecard(input, params, paths?)`. Sem `paths`, roda uma simulação com preços parados; com `paths`, roda uma por caminho e devolve a média. As notas:
  - **`investedPct`:** gasto ÷ aporte, por carteira e no total.
  - **`stability`:** no primeiro aporte, 100 cenários com ruído log-normal de σ = 1%, independente por ativo, com semente fixa 20261031. É a média de `½ Σ |total_i(ruído) − total_i(base)| ÷ aporte`.
  - **`outsideMethod`:** conta as ordens de FII com `price ≥ ceiling`, `bias = "AGUARDAR"` ou `isDividendSuspended(ativo, options)` (Task 3), e as de ação com `price > ceiling` ou `AGUARDAR`.
  - **`fiiDeviation`:** `½ Σ |valor_i ÷ total − weight_i|` dos FIIs, no fim (`final`) e na média dos períodos (`average`).
  - **`fiisFunded`:** quantos FIIs receberam ao menos uma ordem.
  - **`macro`:** por macro, o % do valor da estratégia no fim contra a soma dos `weight` da macro.
  - **`fiiOrders`:** ordens de FII por período e a menor ordem (`null` sem ordens).
  - **`rankWeighted`:** `Σ(rank × total) ÷ Σ total` por classe, sem os de rank nulo (`null` sem ordens).
  - **`brakedShare`:** soma dos `total` de FII com `tensors.dividends < 1` ou `tensors.vacancy < 1`, dividida pelo gasto em FIIs.
  - **`stockIdle`:** sobra média das ações por período, em R$ e em % da parte das ações.
  - **`stockOrders`:** ordens de ação por período e `Σ(quantity ÷ order.lot)` por período.
- [ ] **`index.ts`** exporta os tipos e as funções novas.
- [ ] **Testes**, com carteiras pequenas de números redondos:

  2. `randomPricePaths` dá os mesmos fatores com a mesma semente e fatores diferentes com outra;
  3. `investedPct` bate com um caso conhecido;
