**Goal:** Mostrar o P/VP de cada FII (Carteira Alvo, Recomendações) e avisar quando um FII de papel ou FoF tiver ágio acima de 5%, sem mudar a cascata Suno.

**Architecture:**
- **Origem do VP:** o valor patrimonial por cota vem do Informe Mensal da CVM (FII em zip anual, Fiagro em zip mensal).
- **Ligação ticker ↔ CNPJ:** pela lista oficial de fundos listados da B3 (nome normalizado + ISIN), com checagem de consistência.
- **Atualização:** o cron diário grava o VP no ativo, e o P/VP é calculado na hora, na montagem da carteira.

**Tech Stack:** Next.js 16 (App Router), Mongoose 9, `fflate` (zip), `fetch` com `AbortSignal.timeout`, Tailwind + shadcn (Base UI).

**Spec:** `docs/superpowers/specs/2026-10-18-pvp-design.md`.

**Sem testes automatizados** (decisão do usuário). Cada tarefa se verifica com `npx tsc --noEmit` / `npm run lint`, e a Tarefa 6 confere tudo com os dados reais num script descartável.

// Valor patrimonial (VP) por cota e P/VP dos FIIs, a partir do Informe Mensal da CVM.
// Regras: docs/superpowers/specs/2026-10-18-pvp-design.md (§4 e §5).

/** Um informe mensal de um fundo (FII ou Fiagro), lido do CSV da CVM. */
export type FundReport = {
  cnpj: string;
  /** Mês de referência, AAAA-MM. */
  month: string;
  name: string;
  isin: string | null;
  perShare: number | null;
  netWorth: number | null;
  shares: number | null;
};

/** Um fundo da lista oficial da B3: sigla de negociação e nome oficial. */
export type ListedFund = { acronym: string; fundName: string };

export type FundNav = { perShare: number; month: string; cnpj: string };

/** P/VP fora desta faixa é tratado como dado suspeito e não aparece. */
export const PRICE_TO_NAV_RANGE = { min: 0.3, max: 3 } as const;
const NAV_TOLERANCE = 0.01;
const GENERIC_WORDS = new Set(["FII", "FIAGRO", "DE", "DO", "DA", "DOS", "DAS", "E"]);

/** Nome comparável entre B3 e CVM: sem acento, pontuação nem termos que variam entre as fontes. */
export function normalizeFundName(name: string): string {
  return name
    .normalize("NFD")
    .replace(/\p{Diacritic}/gu, "")
    .toUpperCase()
    .replace(/FUNDO DE INVESTIMENTO NAS CADEIAS PRODUTIVAS AGROINDUSTRIAIS/g, "FIAGRO")
    .replace(/FUNDO DE INVESTIMENTO IMOBILIARIO/g, "FII")
    .replace(/FIAGRO\s*-?\s*IMOBILIARIO/g, "FIAGRO")
    .replace(/RESPONSABILIDADE LIMITADA/g, " ")
    .replace(/CLASSE UNICA/g, " ")
    .replace(/[^A-Z0-9 ]/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}

function wordOverlap(reference: string, candidate: string): number {
  const words = reference.split(" ").filter((word) => word && !GENERIC_WORDS.has(word));
  if (words.length === 0) return 0;
  const other = new Set(candidate.split(" "));
  return words.filter((word) => other.has(word)).length / words.length;
}

/**
 * Liga cada ticker (sigla + 11) ao CNPJ do fundo na CVM (§4). O ISIN sozinho não basta:
 * há ISIN de um fundo com a sigla de outro. Sem ligação segura, o ticker fica sem P/VP.
 */
export function linkFunds(
  listed: readonly ListedFund[],
  reports: readonly FundReport[],
): Map<string, string> {
  const funds = new Map<string, { name: string; isin: string | null }>();
  for (const report of [...reports].sort((a, b) => a.month.localeCompare(b.month))) {
    funds.set(report.cnpj, { name: normalizeFundName(report.name), isin: report.isin });
  }
  const byName = new Map<string, string[]>();
  const byIsinRoot = new Map<string, string[]>();
  for (const [cnpj, fund] of funds) {
    byName.set(fund.name, [...(byName.get(fund.name) ?? []), cnpj]);
    const root = /^BR([A-Z0-9]{4})CTF/.exec(fund.isin ?? "")?.[1];
    if (root) byIsinRoot.set(root, [...(byIsinRoot.get(root) ?? []), cnpj]);
  }

  const links = new Map<string, string>();
  for (const fund of listed) {
    const acronym = fund.acronym.trim().toUpperCase();
    if (!/^[A-Z0-9]{4}$/.test(acronym)) continue;
    const name = normalizeFundName(fund.fundName);
    const byIsin = byIsinRoot.get(acronym) ?? [];
    const named = byName.get(name) ?? [];
    const both = byIsin.filter((cnpj) => named.includes(cnpj));
    const onlyIsin = byIsin.length === 1 ? byIsin[0]! : null;
    const chosen =
      both.length === 1
        ? both[0]!
        : named.length === 1
          ? named[0]!
          : onlyIsin && wordOverlap(name, funds.get(onlyIsin)!.name) >= 0.5
            ? onlyIsin
            : null;
    if (chosen) links.set(`${acronym}11`, chosen);
  }
  return links;
}

function isConsistent(report: FundReport): boolean {
  const { perShare, netWorth, shares } = report;
  if (!perShare || perShare <= 0) return false;
  if (!netWorth || !shares || shares <= 0) return true;
  return Math.abs(perShare - netWorth / shares) <= perShare * NAV_TOLERANCE;
}

/** Último VP por cota de cada CNPJ, só de informes consistentes (§5). */
export function latestNavByCnpj(reports: readonly FundReport[]): Map<string, FundNav> {
  const latest = new Map<string, FundNav>();
  for (const report of reports) {
    if (!isConsistent(report)) continue;
    const current = latest.get(report.cnpj);
    if (!current || report.month > current.month) {
      latest.set(report.cnpj, { perShare: report.perShare!, month: report.month, cnpj: report.cnpj });
    }
  }
  return latest;
}

/** P/VP = cotação ÷ VP por cota; null quando falta dado ou o resultado é suspeito. */
export function priceToNav(
  price: number | null,
  nav: { perShare: number } | null | undefined,
): number | null {
  if (price === null || !nav || nav.perShare <= 0) return null;
  const ratio = price / nav.perShare;
  return ratio >= PRICE_TO_NAV_RANGE.min && ratio <= PRICE_TO_NAV_RANGE.max ? ratio : null;
}

import "server-only";
import { unzipSync } from "fflate";
import type { FundReport } from "../domain/nav";

// Informe Mensal da CVM (dados abertos): FII em zip anual, Fiagro em zip mensal.
// CSV em Latin-1, separado por ";", cabeçalho na primeira linha, decimal com ponto.

const BASE_URL = "https://dados.cvm.gov.br/dados";
const TIMEOUT_MS = 60_000;
const MAX_BYTES = 30 * 1024 * 1024;

async function download(url: string): Promise<Uint8Array | null> {
  const response = await fetch(url, { signal: AbortSignal.timeout(TIMEOUT_MS), cache: "no-store" });
  if (response.status === 404) return null; // mês ainda sem informe publicado
  if (!response.ok) throw new Error(`CVM respondeu ${response.status}`);
  if (Number(response.headers.get("content-length") ?? 0) > MAX_BYTES) {
    throw new Error("arquivo da CVM maior que o limite");
  }
  const bytes = new Uint8Array(await response.arrayBuffer());
  if (bytes.byteLength > MAX_BYTES) throw new Error("arquivo da CVM maior que o limite");
  return bytes;
}

/** Linhas do CSV `<name>.csv` de dentro do zip. */
function readCsv(zip: Uint8Array, name: string): string[][] {
  const files = unzipSync(zip, { filter: (file) => file.name === `${name}.csv` });
  const content = files[`${name}.csv`];
  if (!content) return [];
  return new TextDecoder("latin1")
    .decode(content)
    .split(/\r?\n/)
    .filter(Boolean)
    .map((line) => line.split(";"));
}

function columns<K extends string>(header: readonly string[], names: Record<K, string>) {
  const result = {} as Record<K, number>;
  for (const key of Object.keys(names) as K[]) {
    const index = header.indexOf(names[key]);
    if (index < 0) throw new Error(`coluna ${names[key]} ausente no informe da CVM`);
    result[key] = index;
  }
  return result;
}

const toNumber = (text: string | undefined) => {
  if (!text) return null;
  const number = Number(text);
  return Number.isFinite(number) ? number : null;
};
const toMonth = (date: string | undefined) => (date ?? "").slice(0, 7);
const validMonth = (month: string) => /^\d{4}-\d{2}$/.test(month);

/** Informes de FII de um ano (um zip por ano, com geral + complemento). */
export async function fetchFiiReports(year: number): Promise<FundReport[]> {
  const zip = await download(`${BASE_URL}/FII/DOC/INF_MENSAL/DADOS/inf_mensal_fii_${year}.zip`);
  if (!zip) return [];
  const [generalHeader, ...general] = readCsv(zip, `inf_mensal_fii_geral_${year}`);
  const [complementHeader, ...complement] = readCsv(zip, `inf_mensal_fii_complemento_${year}`);
  if (!generalHeader || !complementHeader) return [];
  const g = columns(generalHeader, {
    cnpj: "CNPJ_Fundo_Classe",
    date: "Data_Referencia",
    name: "Nome_Fundo_Classe",
    isin: "Codigo_ISIN",
  });
  const c = columns(complementHeader, {
    cnpj: "CNPJ_Fundo_Classe",
    date: "Data_Referencia",
    netWorth: "Patrimonio_Liquido",
    shares: "Cotas_Emitidas",
    perShare: "Valor_Patrimonial_Cotas",
  });
  const info = new Map(
    general.map((row) => [`${row[g.cnpj]}|${row[g.date]}`, { name: row[g.name] ?? "", isin: row[g.isin] || null }]),
  );
  return complement
    .map((row) => {
      const meta = info.get(`${row[c.cnpj]}|${row[c.date]}`);
      return {
        cnpj: row[c.cnpj] ?? "",
        month: toMonth(row[c.date]),
        name: meta?.name ?? "",
        isin: meta?.isin ?? null,
        perShare: toNumber(row[c.perShare]),
        netWorth: toNumber(row[c.netWorth]),
        shares: toNumber(row[c.shares]),
      };
    })
    .filter((report) => report.cnpj !== "" && validMonth(report.month));
}

/** Informes de Fiagro de um mês (AAAAMM): um zip por mês. */
export async function fetchFiagroReports(yearMonth: string): Promise<FundReport[]> {
  const zip = await download(
    `${BASE_URL}/FIAGRO/DOC/INF_MENSAL/DADOS/inf_mensal_fiagro_${yearMonth}.zip`,
  );
  if (!zip) return [];
  const [header, ...rows] = readCsv(zip, `inf_mensal_fiagro_${yearMonth}`);
  if (!header) return [];
  const c = columns(header, {
    cnpj: "CNPJ_Classe",
    name: "Nome_Classe",
    date: "Data_Referencia",
    isin: "Codigo_ISIN",
    netWorth: "Patrimonio_Liquido",
    shares: "Cotas_Emitidas",
    perShare: "Valor_Patrimonial_Cotas",
  });
  return rows
    .map((row) => ({
      cnpj: row[c.cnpj] ?? "",
      month: toMonth(row[c.date]),
      name: row[c.name] ?? "",
      isin: row[c.isin] || null,
      perShare: toNumber(row[c.perShare]),
      netWorth: toNumber(row[c.netWorth]),
      shares: toNumber(row[c.shares]),
    }))
    .filter((report) => report.cnpj !== "" && validMonth(report.month));
}

import "server-only";
import type { ListedFund } from "../domain/nav";

// Lista oficial de fundos listados da B3 (pública, sem login): sigla de negociação e nome oficial.
const BASE_URL = "https://sistemaswebb3-listados.b3.com.br/fundsListedProxy/Search/GetListFunds/";
const PAGE_SIZE = 120; // maior página que a B3 aceita
const MAX_PAGES = 20;
const TIMEOUT_MS = 30_000;

type Page = { page?: { totalPages?: number }; results?: { acronym?: string; fundName?: string }[] };

export async function fetchListedFunds(type: "FII" | "FIAGRO"): Promise<ListedFund[]> {
  const funds: ListedFund[] = [];
  for (let pageNumber = 1; pageNumber <= MAX_PAGES; pageNumber++) {
    const query = Buffer.from(
      JSON.stringify({ language: "pt-br", typeFund: type, pageNumber, pageSize: PAGE_SIZE }),
    ).toString("base64");
    const response = await fetch(BASE_URL + query, {
      signal: AbortSignal.timeout(TIMEOUT_MS),
      cache: "no-store",
    });
    if (!response.ok) throw new Error(`B3 respondeu ${response.status}`);
    const body = (await response.json()) as Page;
    for (const item of body.results ?? []) {
      if (item.acronym && item.fundName) funds.push({ acronym: item.acronym, fundName: item.fundName });
    }
    if (pageNumber >= (body.page?.totalPages ?? 0)) break;
  }
  return funds;
}

import { fetchListedFunds } from "../adapters/b3-funds";
import { fetchFiagroReports, fetchFiiReports } from "../adapters/cvm";
import { latestNavByCnpj, linkFunds, type FundReport, type ListedFund } from "../domain/nav";
import { AssetModel } from "./models";

const errorMessage = (error: unknown) => (error instanceof Error ? error.message : String(error));

/** Os últimos `count` meses no formato AAAAMM, do mais recente para o mais antigo. */
function recentYearMonths(now: Date, count: number): string[] {
  return Array.from({ length: count }, (_, back) => {
    const date = new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth() - back, 1));
    return `${date.getUTCFullYear()}${String(date.getUTCMonth() + 1).padStart(2, "0")}`;
  });
}

async function settle<T>(label: string, task: Promise<T>, fallback: T): Promise<T> {
  try {
    return await task;
  } catch (error) {
    console.warn(`[P/VP] ${label}:`, errorMessage(error));
    return fallback;
  }
}

/**
 * Atualiza o VP por cota (Informe Mensal da CVM) dos FIIs pedidos, ligando ticker e CNPJ
 * pela lista de fundos listados da B3. Se uma fonte falhar, o VP anterior continua valendo.
 */
export async function refreshNavs(
  tickers: readonly string[],
  now = new Date(),
): Promise<{ updated: number; unlinked: string[] }> {
  const wanted = [...new Set(tickers)].filter((ticker) => /^[A-Z0-9]{4}11$/.test(ticker));
  if (wanted.length === 0) return { updated: 0, unlinked: [] };
  const year = now.getUTCFullYear();
  const years = now.getUTCMonth() < 2 ? [year - 1, year] : [year];

  const [listed, reports] = await Promise.all([
    Promise.all([
      settle<ListedFund[]>("lista de FIIs da B3", fetchListedFunds("FII"), []),
      settle<ListedFund[]>("lista de Fiagros da B3", fetchListedFunds("FIAGRO"), []),
    ]).then((lists) => lists.flat()),
    Promise.all([
      ...years.map((item) => settle<FundReport[]>(`informe FII ${item}`, fetchFiiReports(item), [])),
      ...recentYearMonths(now, 3).map((month) =>
        settle<FundReport[]>(`informe Fiagro ${month}`, fetchFiagroReports(month), []),
      ),
    ]).then((lists) => lists.flat()),
  ]);

  const links = linkFunds(listed, reports);
  const navs = latestNavByCnpj(reports);
  const updates = wanted.flatMap((ticker) => {
    const cnpj = links.get(ticker);
    const nav = cnpj ? navs.get(cnpj) : undefined;
    return nav
      ? [{
          updateOne: {
            filter: { _id: ticker },
            update: { $set: { nav: { ...nav, fetchedAt: now } } },
            upsert: true,
          },
        }]
